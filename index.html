function doPost(e) { if (e.parameter.action === 'exportExcel') { const manager = e.parameter.manager || ""; const contractor = e.parameter.contractor || ""; const phone = e.parameter.phone || ""; const items = JSON.parse(e.parameter.items || "[]"); const rows = []; rows.push(["Категория", "Номенклатура", "Артикул", "Штрихкод", "Цена", "Количество", "Сумма"]); let total = 0; items.forEach(item => { const sum = item.price * item.qty; total += sum; rows.push([ item.cat, item.name, item.article, item.barcode, item.price, item.qty, sum ]); }); rows.push(["", "", "", "", "", "ИТОГО:", total]); rows.push(["", "", "", "", "", "Менеджер:", manager]); rows.push(["", "", "", "", "", "Контрагент:", contractor]); rows.push(["", "", "", "", "", "Телефон:", phone]); // Разделитель ";" для русской версии Excel const csv = rows.map(row => row.map(cell => { if (cell === undefined || cell === null) return ""; let str = String(cell); if (str.includes('"') || str.includes(';') || str.includes('\n') || str.includes('\r')) { str = '"' + str.replace(/"/g, '""') + '"'; } return str; }).join(";") ).join("\n"); const bom = "\uFEFF"; const output = ContentService.createTextOutput(bom + csv); output.setMimeType(ContentService.MimeType.CSV); return output; } // === Telegram === // Токен и chatId больше не хранятся в коде — они читаются из Script Properties. // Как задать (один раз, вручную): // 1. В редакторе Apps Script: значок шестерёнки слева -> "Свойства проекта" -> "Свойства скрипта" // 2. Добавить два свойства: BOT_TOKEN = <новый токен от @BotFather>, CHAT_ID = <ваш chat id> // Старый токен из этого файла нужно считать скомпрометированным и отозвать через @BotFather -> /revoke. const props = PropertiesService.getScriptProperties(); const token = props.getProperty('BOT_TOKEN'); const chatId = props.getProperty('CHAT_ID'); if (!token || !chatId) { return ContentService.createTextOutput("Ошибка: не заданы BOT_TOKEN/CHAT_ID в свойствах скрипта"); } const text = e.parameter.text; const fileData = e.parameter.fileData; if (!text || text.trim() === "." || text.trim() === "") { return ContentService.createTextOutput("OK"); } const textUrl = `https://api.telegram.org/bot${token}/sendMessage`; UrlFetchApp.fetch(textUrl, { method: "post", payload: { chat_id: chatId, text: text, parse_mode: "HTML" } }); if (fileData) { const fileUrl = `https://api.telegram.org/bot${token}/sendDocument`; const blob = Utilities.newBlob( Utilities.base64Decode(fileData), "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet", "order.xlsx" ); UrlFetchApp.fetch(fileUrl, { method: "post", payload: { chat_id: chatId, document: blob } }); } return ContentService.createTextOutput("OK"); } function doGet() { return ContentService.createTextOutput("Бот работает"); }
